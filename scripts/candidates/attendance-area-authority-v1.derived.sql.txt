-- A null sector is the exact unsectored area. There are no wildcards/company
-- scopes and no inheritance from similarly named leave capabilities or roles.
CREATE TABLE public.attendance_area_scope(
 id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
 tenant_id uuid NOT NULL,
 membership_id uuid NOT NULL,
 source_binding_id uuid NOT NULL,
 capability_key varchar(96) NOT NULL REFERENCES public.iam_capability(capability_key),
 organization_unit_source_id text NOT NULL CHECK(organization_unit_source_id~'^(0|[1-9][0-9]{0,18})$'),
 sector_source_id text CHECK(sector_source_id IS NULL OR sector_source_id~'^(0|[1-9][0-9]{0,18})$'),
 active boolean NOT NULL,
 version integer NOT NULL CHECK(version>0),
 authorization_reference text NOT NULL CHECK(length(btrim(authorization_reference)) BETWEEN 10 AND 500),
 updated_by_user_email text NOT NULL REFERENCES public.internal_users(email),
 updated_at timestamptz NOT NULL DEFAULT clock_timestamp(),
 FOREIGN KEY(membership_id,tenant_id) REFERENCES public.tenant_membership(id,tenant_id),
 FOREIGN KEY(tenant_id,source_binding_id) REFERENCES public.platform_tenant_source_binding(tenant_id,id),
 CHECK(capability_key IN('attendance.area.read','attendance.area.devices.propose','attendance.area.devices.approve','attendance.area.devices.audit.read'))
);
CREATE UNIQUE INDEX attendance_area_scope_identity_v1 ON public.attendance_area_scope
 (tenant_id,membership_id,source_binding_id,capability_key,organization_unit_source_id,coalesce(sector_source_id,''));
CREATE TABLE public.attendance_area_scope_event(
 id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
 scope_id uuid NOT NULL REFERENCES public.attendance_area_scope(id),
 version integer NOT NULL,
 before_value jsonb,
 after_value jsonb NOT NULL,
 recorded_at timestamptz NOT NULL DEFAULT clock_timestamp(),
 UNIQUE(scope_id,version)
);
CREATE FUNCTION public.attendance_area_scope_guard_v1() RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
BEGIN
 IF TG_OP IN('DELETE','TRUNCATE') THEN RAISE EXCEPTION 'ATTENDANCE_AREA_SCOPE_IMMUTABLE'; END IF;
 IF TG_OP='INSERT' AND NEW.version<>1 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_SCOPE_VERSION_CONFLICT'; END IF;
 IF TG_OP='UPDATE' AND (NEW.id IS DISTINCT FROM OLD.id OR NEW.tenant_id IS DISTINCT FROM OLD.tenant_id
  OR NEW.membership_id IS DISTINCT FROM OLD.membership_id OR NEW.source_binding_id IS DISTINCT FROM OLD.source_binding_id
  OR NEW.capability_key IS DISTINCT FROM OLD.capability_key OR NEW.organization_unit_source_id IS DISTINCT FROM OLD.organization_unit_source_id
  OR NEW.sector_source_id IS DISTINCT FROM OLD.sector_source_id OR NEW.version<>OLD.version+1)
 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_SCOPE_VERSION_CONFLICT'; END IF;
 NEW.updated_at:=clock_timestamp(); RETURN NEW;
END $$;

CREATE FUNCTION public.attendance_area_catalog_label_v1(catalog jsonb,area jsonb) RETURNS jsonb
LANGUAGE plpgsql IMMUTABLE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE org_label text; sector_label text; matches bigint; label_value text;
BEGIN
 IF jsonb_typeof(area) IS DISTINCT FROM 'object' OR (SELECT count(*) FROM jsonb_object_keys(area))<>2
  OR NOT area ?& ARRAY['organizationUnitSourceId','sectorSourceId']
  OR jsonb_typeof(area->'organizationUnitSourceId') IS DISTINCT FROM 'string'
  OR area->>'organizationUnitSourceId'!~'^(0|[1-9][0-9]{0,18})$'
  OR (area->'sectorSourceId'<>'null'::jsonb AND (jsonb_typeof(area->'sectorSourceId') IS DISTINCT FROM 'string' OR area->>'sectorSourceId'!~'^(0|[1-9][0-9]{0,18})$'))
  OR jsonb_typeof(catalog->'items') IS DISTINCT FROM 'array'
 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_QUERY_INVALID'; END IF;
 SELECT count(*),min(item->>'label') INTO matches,org_label FROM jsonb_array_elements(catalog->'items') item
 WHERE item->>'kind'='organizations' AND item->>'code'=area->>'organizationUnitSourceId';
 IF matches<>1 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_CATALOG_CHANGED'; END IF;
 IF area->>'sectorSourceId' IS NOT NULL THEN
  SELECT count(*),min(item->>'label') INTO matches,sector_label FROM jsonb_array_elements(catalog->'items') item
  WHERE item->>'kind'='sectors' AND item->>'code'=area->>'sectorSourceId';
  IF matches<>1 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_CATALOG_CHANGED'; END IF;
 END IF;
 -- The catalogue names each component; it does not itself assert that a sector
 -- belongs to an organization. The explicit grant and historical source do that.
 label_value:=org_label||CASE WHEN sector_label IS NULL THEN ' / Sin sector' ELSE ' / '||sector_label END;
 IF org_label IS NULL OR length(btrim(org_label))=0 OR (area->>'sectorSourceId' IS NOT NULL AND length(btrim(coalesce(sector_label,'')))=0)
  OR length(label_value) NOT BETWEEN 1 AND 500 OR label_value~'[<>[:cntrl:]]' OR label_value<>normalize(label_value,NFC)
  OR label_value~U&'[\00AD\0600-\0605\061C\06DD\070F\0890-\0891\08E2\180E\200B-\200F\202A-\202E\2060-\2064\2066-\206F\FEFF\FFF9-\FFFB\+0110BD\+0110CD\+013430-\+01343F\+01BCA0-\+01BCA3\+01D173-\+01D17A\+0E0001\+0E0020-\+0E007F]'
 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_CATALOG_CHANGED'; END IF;
 RETURN area||jsonb_build_object('areaLabel',label_value);
END $$;

-- Private helper. Only externally authenticated facades may call this. Every
-- operation rechecks the real session, certified release/binding and live IAM.
CREATE FUNCTION public.attendance_area_context_v1(p jsonb,required_capability text) RETURNS jsonb
LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE c jsonb; catalog jsonb; capabilities jsonb; all_scopes jsonb:='[]'; scopes jsonb:='[]';
 s public.attendance_area_scope%ROWTYPE; named_area jsonb; scope_value jsonb; actor_id uuid; fingerprint text;
BEGIN
 IF jsonb_typeof(p) IS DISTINCT FROM 'object' OR NOT p ?& ARRAY['actorEmail','actorSessionId','actorSessionVersion','releaseSha','tenantId','membershipId']
  OR (SELECT count(*) FROM jsonb_object_keys(p))<>6 OR required_capability IS NULL OR required_capability NOT IN('attendance.area.read','attendance.area.devices.propose','attendance.area.devices.approve','attendance.area.devices.audit.read')
 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_SESSION_INVALID'; END IF;
 c:=public.attendance_gateway_assert_session_v1(p->>'actorEmail',(p->>'actorSessionId')::uuid,(p->>'actorSessionVersion')::integer,p->>'releaseSha',(p->>'tenantId')::uuid,(p->>'membershipId')::uuid);
 LOCK TABLE public.iam_role_capability,public.tenant_membership_capability_override,public.attendance_area_scope,
  public.native_employment_catalog_proposal,public.native_employment_catalog_review IN SHARE MODE NOWAIT;
 PERFORM public.tenant_iam_assert_no_sod_conflict((c->>'membershipId')::uuid);
 SELECT coalesce(jsonb_agg(capability_key ORDER BY capability_key),'[]'::jsonb) INTO capabilities
 FROM public.tenant_iam_effective_capabilities((c->>'membershipId')::uuid)
 WHERE capability_key IN('attendance.area.read','attendance.area.devices.propose','attendance.area.devices.approve','attendance.area.devices.audit.read');
 IF NOT capabilities ? required_capability OR NOT capabilities ? 'attendance.area.read' THEN RAISE EXCEPTION 'ATTENDANCE_AREA_CAPABILITY_REQUIRED'; END IF;
 c:=c||jsonb_build_object('sourceBindingId',c->>'certifiedBindingId','sourceCompanyId',c->>'sourceCompanyId',
  'actorEmail',p->>'actorEmail','actorSessionId',p->>'actorSessionId','actorSessionVersion',(p->>'actorSessionVersion')::integer,'releaseSha',p->>'releaseSha');
 BEGIN
  actor_id:=public.time_catalog_native_actor_v2((c->>'tenantId')::uuid,(c->>'sourceBindingId')::uuid,(c->>'membershipId')::uuid);
 EXCEPTION WHEN SQLSTATE 'P0001' THEN
  IF SQLERRM IN('TIME_CATALOG_EMPLOYMENT_REQUIRED','TIME_CATALOG_ASSIGNMENT_CONTRACT_INVALID','TIME_CATALOG_NATIVE_PERIOD_INVALID')
   AND required_capability IN('attendance.area.read','attendance.area.devices.audit.read') THEN actor_id:=NULL;
  ELSE RAISE; END IF;
 END;
 IF required_capability IN('attendance.area.devices.propose','attendance.area.devices.approve') AND actor_id IS NULL THEN RAISE EXCEPTION 'ATTENDANCE_AREA_ACTOR_REQUIRED'; END IF;
 PERFORM public.grh_curated_source_read_lock_v1();
 IF EXISTS(SELECT 1 FROM public.attendance_area_scope a WHERE a.tenant_id=(c->>'tenantId')::uuid AND a.membership_id=(c->>'membershipId')::uuid
  AND a.source_binding_id=(c->>'sourceBindingId')::uuid AND a.active AND capabilities ? a.capability_key::text) THEN
  catalog:=public.native_employee_catalog_v1(c);
  IF catalog->>'version' IS NULL OR catalog->>'version'!~'^[a-f0-9]{64}$' THEN RAISE EXCEPTION 'ATTENDANCE_AREA_CATALOG_CHANGED'; END IF;
 ELSE catalog:=jsonb_build_object('items','[]'::jsonb,'version',repeat('0',64)); END IF;
 FOR s IN SELECT * FROM public.attendance_area_scope a WHERE a.tenant_id=(c->>'tenantId')::uuid AND a.membership_id=(c->>'membershipId')::uuid
  AND a.source_binding_id=(c->>'sourceBindingId')::uuid AND a.active AND capabilities ? a.capability_key::text ORDER BY a.id LOOP
  named_area:=public.attendance_area_catalog_label_v1(catalog,jsonb_build_object('organizationUnitSourceId',s.organization_unit_source_id,'sectorSourceId',s.sector_source_id));
  scope_value:=named_area||jsonb_build_object('scopeId',s.id,'version',s.version);
  all_scopes:=all_scopes||jsonb_build_array(scope_value||jsonb_build_object('capability',s.capability_key));
  IF s.capability_key=required_capability THEN scopes:=scopes||jsonb_build_array(scope_value); END IF;
 END LOOP;
 IF jsonb_array_length(all_scopes)>2000 OR jsonb_array_length(scopes)>500 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_SCOPE_LIMIT'; END IF;
 -- Hash is an optimistic concurrency token, not authority. It binds session,
 -- person, every effective grant and catalogue labels; all are reread above.
 fingerprint:=encode(public.digest(jsonb_build_object('session',p,'binding',c->>'sourceBindingId','company',c->>'sourceCompanyId','database',c->>'sourceDatabase',
  'actor',actor_id,'capabilities',capabilities,'scopes',all_scopes,'catalogVersion',catalog->>'version')::text,'sha256'),'hex');
 RETURN c||jsonb_build_object('actorPersonId',actor_id,'capabilities',capabilities,'areaScopes',scopes,'scopeVersion',fingerprint,'areaCatalogVersion',catalog->>'version');
EXCEPTION WHEN lock_not_available THEN RAISE EXCEPTION 'ATTENDANCE_AREA_BUSY';
END $$;

CREATE FUNCTION public.attendance_area_assert_scope_v1(c jsonb,area jsonb) RETURNS jsonb
LANGUAGE plpgsql IMMUTABLE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE selected jsonb; matches bigint;
BEGIN
 IF jsonb_typeof(area) IS DISTINCT FROM 'object' OR NOT area ?& ARRAY['organizationUnitSourceId','sectorSourceId']
  OR (SELECT count(*) FROM jsonb_object_keys(area))<>2 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_QUERY_INVALID'; END IF;
 SELECT count(*),jsonb_agg(s)->0 INTO matches,selected FROM jsonb_array_elements(c->'areaScopes') s
 WHERE s->'organizationUnitSourceId'=area->'organizationUnitSourceId' AND s->'sectorSourceId'=area->'sectorSourceId';
 IF matches<>1 THEN RAISE EXCEPTION 'ATTENDANCE_AREA_SCOPE_DENIED'; END IF;
 RETURN area||jsonb_build_object('areaLabel',selected->>'areaLabel');
END $$;
