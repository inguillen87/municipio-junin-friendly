CREATE FUNCTION public.own_employment_print_provenance_candidate_v1(proof jsonb,registration_value jsonb)
RETURNS void LANGUAGE plpgsql IMMUTABLE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $proof$
DECLARE final_value jsonb;field text;BEGIN
 IF proof->>'kind'='native_registration' THEN
  IF public.own_program_exact_v1(proof,ARRAY['kind','registrationId','requestSha256']) IS NOT TRUE THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
 ELSIF proof->>'kind'='employment_adoption_final' THEN
  IF public.own_program_exact_v1(proof,ARRAY['kind','registrationId','requestSha256','adoptionProposalId','adoptionReviewId','sealSha256','finalRevision']) IS NOT TRUE THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  FOREACH field IN ARRAY ARRAY['adoptionProposalId','adoptionReviewId'] LOOP
   IF jsonb_typeof(proof->field) IS DISTINCT FROM 'string' OR proof->>field!~'^[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}$' OR proof->>field='00000000-0000-0000-0000-000000000000' THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  END LOOP;
  IF jsonb_typeof(proof->'sealSha256') IS DISTINCT FROM 'string' OR proof->>'sealSha256'!~'^[a-f0-9]{64}$' THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  final_value:=proof->'finalRevision';
  IF public.own_program_exact_v1(final_value,ARRAY['revisionId','packageSha256','sourceSha256','cutoff','coreManifestSha256','curatedManifestSha256','factsSha256']) IS NOT TRUE
   OR jsonb_typeof(final_value->'revisionId') IS DISTINCT FROM 'string' OR final_value->>'revisionId'!~'^[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}$'
   OR final_value->>'revisionId'='00000000-0000-0000-0000-000000000000' THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  FOREACH field IN ARRAY ARRAY['packageSha256','sourceSha256','coreManifestSha256','curatedManifestSha256','factsSha256'] LOOP
   IF jsonb_typeof(final_value->field) IS DISTINCT FROM 'string' OR final_value->>field!~'^[a-f0-9]{64}$' THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  END LOOP;
  IF jsonb_typeof(final_value->'cutoff') IS DISTINCT FROM 'string' OR final_value->>'cutoff'!~'^[0-9]{4}-[0-9]{2}-[0-9]{2}T([01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9]$'
   OR left(final_value->>'cutoff',4)='0000' OR NOT pg_input_is_valid(left(final_value->>'cutoff',10),'date')
   OR to_char(left(final_value->>'cutoff',10)::date,'YYYY-MM-DD')<>left(final_value->>'cutoff',10) THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
 ELSE RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
 IF proof->'registrationId' IS DISTINCT FROM registration_value OR jsonb_typeof(proof->'registrationId') IS DISTINCT FROM 'string'
  OR proof->>'registrationId'!~'^[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}$' OR proof->>'registrationId'='00000000-0000-0000-0000-000000000000'
  OR jsonb_typeof(proof->'requestSha256') IS DISTINCT FROM 'string' OR proof->>'requestSha256'!~'^[a-f0-9]{64}$' THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
END $proof$;

CREATE FUNCTION public.own_employment_print_component_candidate_v1(component jsonb,classifications jsonb,employees jsonb,period_value text)
RETURNS void LANGUAGE plpgsql IMMUTABLE SECURITY DEFINER SET search_path=pg_catalog,public,pg_temp AS $component$
DECLARE classification jsonb;row_value jsonb;employee jsonb;proof jsonb;n integer;title_value text;
 title_units integer;title_forbidden boolean;title_edge_space boolean;BEGIN
 IF public.own_program_exact_v1(component,ARRAY['version','period','complete','total','classificationsSha256','rows']) IS NOT TRUE
  OR component->>'version' IS DISTINCT FROM 'own-employment-print-capture.v1' OR component->>'period' IS DISTINCT FROM period_value
  OR period_value IS NULL OR period_value!~'^(19|20)[0-9]{2}-(0[1-9]|1[0-2])$'
  OR component->'complete' IS DISTINCT FROM 'true'::jsonb OR jsonb_typeof(component->'rows') IS DISTINCT FROM 'array'
  OR public.own_program_exact_v1(classifications,ARRAY['version','complete','total','rows']) IS NOT TRUE
  OR classifications->>'version' IS DISTINCT FROM 'own-period-classification-capture.v1' OR classifications->'complete' IS DISTINCT FROM 'true'::jsonb
  OR jsonb_typeof(classifications->'rows') IS DISTINCT FROM 'array' OR jsonb_typeof(employees) IS DISTINCT FROM 'array'
 THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
 IF jsonb_array_length(employees)>10000 OR component->'total' IS DISTINCT FROM to_jsonb(jsonb_array_length(employees))
  OR classifications->'total' IS DISTINCT FROM component->'total' OR jsonb_array_length(component->'rows')<>jsonb_array_length(employees)
  OR jsonb_array_length(classifications->'rows')<>jsonb_array_length(employees)
  OR component->>'classificationsSha256' IS DISTINCT FROM public.own_run_hash_v1(classifications)
  OR EXISTS(SELECT 1 FROM jsonb_array_elements(component->'rows') r GROUP BY r->>'contractId' HAVING count(*)<>1)
  OR EXISTS(SELECT 1 FROM jsonb_array_elements(component->'rows') r GROUP BY r->>'registrationId' HAVING count(*)<>1)
  OR EXISTS(SELECT 1 FROM jsonb_array_elements(classifications->'rows') r GROUP BY r->>'contractId' HAVING count(*)<>1)
  OR EXISTS(SELECT 1 FROM jsonb_array_elements(classifications->'rows') r GROUP BY r->>'registrationId' HAVING count(*)<>1)
  OR EXISTS(SELECT 1 FROM jsonb_array_elements(employees) e GROUP BY e->>'contractId' HAVING count(*)<>1)
 THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
 FOR classification IN SELECT value FROM jsonb_array_elements(classifications->'rows') LOOP
  SELECT count(*),jsonb_agg(e)->0 INTO n,employee FROM jsonb_array_elements(employees) e WHERE e->'contractId'=classification->'contractId';
  IF n<>1 THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  PERFORM public.own_period_classification_capture_row_v1(classification,employee,period_value);
  SELECT count(*),jsonb_agg(r)->0 INTO n,row_value FROM jsonb_array_elements(component->'rows') r WHERE r->'contractId'=classification->'contractId';
  IF n<>1 OR public.own_program_exact_v1(row_value,ARRAY['contractId','registrationId','identityToken','classificationRowSha256','basis','jobTitle','provenance']) IS NOT TRUE
   OR row_value->'registrationId' IS DISTINCT FROM classification->'registrationId' OR row_value->'identityToken' IS DISTINCT FROM classification->'identityToken'
   OR row_value->'basis' IS DISTINCT FROM classification->'basis' OR row_value->>'classificationRowSha256' IS DISTINCT FROM public.own_run_hash_v1(classification)
  THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  IF row_value->'jobTitle' IS DISTINCT FROM 'null'::jsonb THEN
   IF jsonb_typeof(row_value->'jobTitle') IS DISTINCT FROM 'string' THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
   title_value:=row_value->>'jobTitle';
   -- Exact JS contract, independent of locale-sensitive regex classes and the
   -- original receipt helper's ordinary-space btrim. Nothing is normalized.
   -- PostgreSQL UTF8 text/jsonb already rejects NUL and unpaired surrogates.
   SELECT coalesce(sum(CASE WHEN cp>65535 THEN 2 ELSE 1 END),0),
    coalesce(bool_or(cp BETWEEN 0 AND 31 OR cp BETWEEN 127 AND 159 OR cp IN(60,62)
      OR cp BETWEEN 8234 AND 8238 OR cp BETWEEN 8294 AND 8297),false),
    coalesce(bool_or((ordinal_value=1 OR ordinal_value=char_length(title_value)) AND
      (cp IN(9,10,11,12,13,32,160,5760,8232,8233,8239,8287,12288,65279) OR cp BETWEEN 8192 AND 8202)),false)
    INTO title_units,title_forbidden,title_edge_space
    FROM(SELECT i AS ordinal_value,ascii(substr(title_value,i,1)) AS cp FROM generate_series(1,char_length(title_value)) i) characters;
   IF title_units NOT BETWEEN 1 AND 120 OR title_forbidden OR title_edge_space
    OR normalize(title_value,NFC) IS DISTINCT FROM title_value
   THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  END IF;
  proof:=row_value->'provenance';
  IF row_value->>'basis'='current_own_registration' THEN
   IF public.own_program_exact_v1(proof,ARRAY['kind','sourceField','sourceVersion','registration']) IS NOT TRUE
    OR proof->>'kind' IS DISTINCT FROM 'current_own_contract_at_capture' OR proof->>'sourceField' IS DISTINCT FROM 'source_payload.employment.cargoName'
    OR jsonb_typeof(proof->'sourceVersion') IS DISTINCT FROM 'string' OR proof->>'sourceVersion'!~'^[a-f0-9]{64}$'
   THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
   PERFORM public.own_employment_print_provenance_candidate_v1(proof->'registration',row_value->'registrationId');
  ELSIF row_value->>'basis'='approved_own_period_classification' THEN
   IF row_value->'jobTitle' IS DISTINCT FROM 'null'::jsonb OR proof IS DISTINCT FROM classification->'provenance'
   THEN RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
  ELSE RAISE EXCEPTION 'OWN_RUN_EMPLOYMENT_PRINT_INVALID';END IF;
 END LOOP;
END $component$;

CREATE OR REPLACE FUNCTION public.own_run_capture_v1(p jsonb, body jsonb, key uuid, algorithm text)

  inventory:=inventory||jsonb_build_object('familyPayrollFacts',jsonb_build_object('version',family_facts->>'version','total',jsonb_array_length(family_facts->'records'),'complete',true,'authority','approved_family_declaration'));
  payload_value:=jsonb_set(jsonb_set(payload_value,'{population}',population),'{sourceInventory}',inventory);
  PERFORM public.own_run_family_expected_inputs_v1(payload_value);
 END IF;
 IF program_state#>>'{program,definition,familyTariffsVersion}' IN('own-payroll-family-tariffs.v2','own-payroll-family-tariffs.v3','own-payroll-family-tariffs.v4') THEN
  inventory:=inventory||jsonb_build_object('familyTariffs',public.own_run_family_tariff_native_capture_candidate_v2(payload_value));
  payload_value:=jsonb_set(payload_value,'{sourceInventory}',inventory);
 END IF;
 IF emission_enabled THEN
  inventory:=inventory||jsonb_build_object('rowEmission',public.own_run_row_emission_inventory_candidate_v1(payload_value,official_family));
  payload_value:=jsonb_set(payload_value,'{sourceInventory}',inventory);
 END IF;
 PERFORM public.own_run_catalogue_values_expected_inputs_v1(payload_value);
 PERFORM public.own_run_source_coverage_check_v1(payload_value);
 IF program_state#>>'{program,definition,monthlyDeclaredAdjustmentsVersion}'='own-monthly-declared-adjustments.v1' THEN
  payload_value:=jsonb_set(payload_value,'{monthly,nativeBatches}',native_batches);
  payload_value:=jsonb_set(payload_value,'{sourceInventory,declaredAdjustments}',public.own_run_declared_adjustments_capture_v1(payload_value));
 END IF;
 IF program_state#>>'{program,definition,optionalScalesVersion}'='own-payroll-optional-scales.v1' THEN PERFORM public.own_run_optional_scale_expected_inputs_v1(payload_value);END IF;
 PERFORM public.own_run_quantity_only_expected_inputs_v1(payload_value);
 IF program_state#>>'{program,definition,conditionalSourcesVersion}'='own-payroll-conditional-sources.v1' THEN PERFORM public.own_run_conditional_expected_inputs_v1(payload_value);END IF;
 IF program_state#>>'{program,definition,cappedDeductionsVersion}'='own-capped-deductions.v1' THEN payload_value:=jsonb_set(payload_value,'{sourceInventory,cappedDeductions}',public.own_run_capped_inventory_v1(payload_value));END IF;
 PERFORM public.own_run_capped_verify_v1(payload_value,NULL,NULL);
 IF octet_length(payload_value::text)+octet_length(body::text)+8192>4194304 OR (SELECT count(*) FROM public.own_payroll_run_capture WHERE tenant_id=(ctx->>'tenantId')::uuid AND source_binding_id=(ctx->>'sourceBindingId')::uuid)>=1000 THEN RAISE EXCEPTION 'OWN_RUN_LIMIT';END IF;
 PERFORM public.own_run_capacity_v1(octet_length(payload_value::text)+octet_length(body::text));
 PERFORM public.own_close_guard_v1(ctx,body->>'period',body->>'liquidationType',ids);
 INSERT INTO public.own_payroll_run_capture(tenant_id,source_binding_id,actor_membership_id,actor_person_id,actor_email,actor_session_id,actor_session_version,release_sha,request_key,body,body_sha256,algorithm_sha256,payload,payload_sha256) VALUES((ctx->>'tenantId')::uuid,(ctx->>'sourceBindingId')::uuid,(ctx->>'membershipId')::uuid,(ctx->>'actorPersonId')::uuid,ctx->>'actorEmail',(p->>'actorSessionId')::uuid,(p->>'actorSessionVersion')::integer,p->>'releaseSha',key,body,fingerprint,algorithm,payload_value,public.own_run_hash_v1(payload_value)) RETURNING * INTO c;
 RETURN public.own_run_receipt_v1(c,false);END $function$;

CREATE OR REPLACE FUNCTION public.own_receipt_print_facts_v6_candidate_v1(record_value jsonb,capture jsonb,bank_fact jsonb,frozen_at text)
RETURNS jsonb LANGUAGE plpgsql IMMUTABLE SECURITY DEFINER
SET search_path=pg_catalog,public,pg_temp AS $v6facts$
DECLARE cid text:=record_value->>'contractId';saved jsonb:=capture->'saved';payload jsonb:=capture->'payload';
 input_value jsonb:=saved->'input';result_value jsonb:=saved->'result';population jsonb;employee jsonb;
 component jsonb;employment_row jsonb;row_value jsonb;other_employee jsonb;definition jsonb;
 units_value jsonb:='[]';concept jsonb;quantity jsonb;captured_quantities jsonb;destination jsonb;
 start_value jsonb:='null';category_value jsonb:='null';employment_hash jsonb:='null';job_title_value jsonb:='null';n integer;
BEGIN
 IF jsonb_typeof(record_value) IS DISTINCT FROM 'object' OR cid IS NULL
 OR cid !~ '^[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}$'
 OR coalesce(record_value->>'runId','') !~ '^[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}$'
 OR coalesce(record_value->>'registrationId','') !~ '^[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}$'
 OR coalesce(input_value->>'period','') !~ '^(19|20)[0-9]{2}-(0[1-9]|1[0-2])$'
 OR coalesce(input_value->>'liquidationType' IN('monthly','first_fortnight','sac','vacation','supplementary','final','other'),false) IS NOT TRUE
 OR jsonb_typeof(record_value->'inputVersion') IS DISTINCT FROM 'string'
 OR coalesce(record_value->>'inputVersion' IN('own-payroll-input.v4','own-payroll-input.v5'),false) IS NOT TRUE
 OR record_value->>'resultVersion' IS DISTINCT FROM replace(record_value->>'inputVersion','input','result')
 OR capture->>'id' IS DISTINCT FROM record_value->>'runId'
 OR jsonb_typeof(payload) IS DISTINCT FROM 'object' OR jsonb_typeof(saved) IS DISTINCT FROM 'object'
 OR jsonb_typeof(input_value) IS DISTINCT FROM 'object' OR jsonb_typeof(result_value) IS DISTINCT FROM 'object'
 OR payload->>'period' IS DISTINCT FROM input_value->>'period'
 OR payload->>'liquidationType' IS DISTINCT FROM input_value->>'liquidationType'
 OR result_value->>'period' IS DISTINCT FROM input_value->>'period'
 OR result_value->>'liquidationType' IS DISTINCT FROM input_value->>'liquidationType'
 OR input_value->>'version' IS DISTINCT FROM record_value->>'inputVersion'
 OR result_value->>'version' IS DISTINCT FROM record_value->>'resultVersion'
 OR capture->'payloadSha256' IS DISTINCT FROM record_value->'payloadSha256'
 OR saved->'inputSha256' IS DISTINCT FROM record_value->'inputSha256'
 OR saved->'resultSha256' IS DISTINCT FROM record_value->'resultSha256'
 OR capture->>'payloadSha256' IS DISTINCT FROM public.own_run_hash_v1(payload)
 OR saved->>'inputSha256' IS DISTINCT FROM public.own_run_hash_v1(input_value)
 OR saved->>'resultSha256' IS DISTINCT FROM public.own_run_hash_v1(result_value)
 OR jsonb_typeof(record_value->'concepts') IS DISTINCT FROM 'array'
 OR jsonb_typeof(record_value->'receiptRowKeys') IS DISTINCT FROM 'array'
 OR jsonb_typeof(record_value->'quantities') IS DISTINCT FROM 'array'
 OR jsonb_typeof(result_value->'quantities') IS DISTINCT FROM 'array'
 OR jsonb_typeof(payload#>'{population,employees}') IS DISTINCT FROM 'array'
 OR frozen_at IS NULL OR frozen_at !~ '^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}\.[0-9]{6}Z$'
 OR NOT pg_input_is_valid(frozen_at,'timestamp with time zone')
 THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 IF to_char(frozen_at::timestamptz AT TIME ZONE 'UTC','YYYY-MM-DD"T"HH24:MI:SS.US"Z"') IS DISTINCT FROM frozen_at
 THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 population:=payload#>'{population,employees}';
 SELECT count(*) INTO n FROM jsonb_array_elements(population) e WHERE e->>'contractId'=cid;
 IF n<>1 THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 SELECT e INTO employee FROM jsonb_array_elements(population) e WHERE e->>'contractId'=cid;
 IF employee->'employeeNumber' IS DISTINCT FROM record_value->'employeeNumber'
 OR employee->'agreementCode' IS DISTINCT FROM record_value->'agreementCode'
 OR employee->'departmentCode' IS DISTINCT FROM record_value->'departmentCode'
 THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 SELECT coalesce(jsonb_agg(q ORDER BY ord),'[]') INTO captured_quantities
 FROM jsonb_array_elements(result_value->'quantities') WITH ORDINALITY source_rows(q,ord) WHERE q->>'contractId'=cid;
 IF captured_quantities IS DISTINCT FROM record_value->'quantities'
 OR EXISTS(SELECT 1 FROM jsonb_array_elements(record_value->'quantities') q GROUP BY q->>'conceptCode' HAVING count(*)<>1)
 OR EXISTS(SELECT 1 FROM jsonb_array_elements(record_value->'concepts') c GROUP BY c->>'code' HAVING count(*)<>1)
 THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 FOR concept IN SELECT value FROM jsonb_array_elements(record_value->'concepts') LOOP
  IF concept->>'nature' IS NULL OR concept->>'nature'='auxiliary' OR concept->>'code' IS NULL
  OR concept->>'code' !~ '^[0-9]{1,9}$' OR NOT record_value->'receiptRowKeys' ? (cid||':concept:'||(concept->>'code'))
  THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
  SELECT q INTO quantity FROM jsonb_array_elements(record_value->'quantities') q WHERE q->>'contractId'=cid AND q->>'conceptCode'=concept->>'code';
  units_value:=units_value||jsonb_build_array(jsonb_build_object('rowKey',cid||':concept:'||(concept->>'code'),'quantity',coalesce(quantity,'null')));
 END LOOP;
 IF jsonb_array_length(units_value)<>jsonb_array_length(record_value->'receiptRowKeys') THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 IF payload->'sourceInventory' ? 'periodClassifications' THEN
  component:=payload#>'{sourceInventory,periodClassifications}';
  IF public.own_program_exact_v1(component,ARRAY['version','complete','total','rows']) IS NOT TRUE
  OR component->>'version' IS DISTINCT FROM 'own-period-classification-capture.v1'
  OR component->'complete' IS DISTINCT FROM 'true'::jsonb OR jsonb_typeof(component->'rows') IS DISTINCT FROM 'array'
  OR component->'total' IS DISTINCT FROM to_jsonb(jsonb_array_length(population))
  OR jsonb_array_length(component->'rows')<>jsonb_array_length(population)
  OR EXISTS(SELECT 1 FROM jsonb_array_elements(component->'rows') x GROUP BY x->>'contractId' HAVING count(*)<>1)
  OR EXISTS(SELECT 1 FROM jsonb_array_elements(component->'rows') x GROUP BY x->>'registrationId' HAVING count(*)<>1)
  THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
  FOR row_value IN SELECT value FROM jsonb_array_elements(component->'rows') LOOP
   SELECT count(*) INTO n FROM jsonb_array_elements(population) e WHERE e->'contractId'=row_value->'contractId';
   IF n<>1 THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
   SELECT e INTO other_employee FROM jsonb_array_elements(population) e WHERE e->'contractId'=row_value->'contractId';
   BEGIN
    PERFORM public.own_period_classification_capture_row_v1(row_value,other_employee,payload->>'period');
   EXCEPTION WHEN OTHERS THEN
    IF SQLERRM='OWN_CLOSE_CONTRACT_INVALID' THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;RAISE;
   END;
   IF row_value->>'contractId'=cid THEN employment_row:=row_value;END IF;
  END LOOP;
  IF employment_row IS NULL OR employment_row->'registrationId' IS DISTINCT FROM record_value->'registrationId'
  THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
  definition:=employment_row->'definition';start_value:=definition->'startDate';category_value:=definition->'categoryCode';
  employment_hash:=to_jsonb(public.own_run_hash_v1(component));
 END IF;
 IF payload->'sourceInventory' ? 'employmentPrintFacts' THEN
  DECLARE print_component jsonb:=payload#>'{sourceInventory,employmentPrintFacts}';
   print_classifications jsonb:=payload#>'{sourceInventory,periodClassifications}';print_row jsonb;
  BEGIN
   PERFORM public.own_employment_print_component_candidate_v1(print_component,print_classifications,population,payload->>'period');
   SELECT x INTO print_row FROM jsonb_array_elements(print_component->'rows') x WHERE x->'contractId'=record_value->'contractId';
   IF print_row IS NULL OR print_row->'registrationId' IS DISTINCT FROM record_value->'registrationId'
   THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
   job_title_value:=print_row->'jobTitle';
   employment_hash:=to_jsonb(public.own_run_hash_v1(jsonb_build_object('version','own-employment-print-evidence.v1',
    'periodClassifications',print_classifications,'employmentPrintFacts',print_component)));
  END;
 END IF;
 IF public.own_program_exact_v1(bank_fact,ARRAY['paymentDestination','bankConfigurationVersion','bankApprovalId']) IS NOT TRUE
 THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 destination:=bank_fact->'paymentDestination';
 IF destination='null'::jsonb THEN
  IF bank_fact->'bankConfigurationVersion' IS DISTINCT FROM 'null'::jsonb OR bank_fact->'bankApprovalId' IS DISTINCT FROM 'null'::jsonb
  THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 ELSE
  IF public.own_program_exact_v1(destination,ARRAY['bankLabel','accountType','accountNumberMasked','currency','referenceDate']) IS NOT TRUE
  OR NOT public.own_receipt_text_v1(destination->'bankLabel',1,160)
  OR (destination->'accountType' IS DISTINCT FROM 'null'::jsonb AND
    (jsonb_typeof(destination->'accountType') IS DISTINCT FROM 'string' OR coalesce(destination->>'accountType' IN('CA','CC'),false) IS NOT TRUE))
  OR jsonb_typeof(destination->'currency') IS DISTINCT FROM 'string'
  OR coalesce(destination->>'currency' IN('ARS','USD'),false) IS NOT TRUE OR NOT public.own_bank_accounts_day_v1(destination->'referenceDate')
  OR (destination->'accountNumberMasked' IS DISTINCT FROM 'null'::jsonb AND
    (NOT public.own_receipt_text_v1(destination->'accountNumberMasked',1,40)
     OR CASE WHEN char_length(destination->>'accountNumberMasked')<=4 THEN
       destination->>'accountNumberMasked' IS DISTINCT FROM repeat('*',char_length(destination->>'accountNumberMasked'))
      ELSE left(destination->>'accountNumberMasked',char_length(destination->>'accountNumberMasked')-4)
       IS DISTINCT FROM repeat('*',char_length(destination->>'accountNumberMasked')-4) END))
  OR jsonb_typeof(bank_fact->'bankConfigurationVersion') IS DISTINCT FROM 'string'
  OR bank_fact->>'bankConfigurationVersion' !~ '^[a-f0-9]{64}$'
  OR jsonb_typeof(bank_fact->'bankApprovalId') IS DISTINCT FROM 'string'
  OR bank_fact->>'bankApprovalId' !~ '^[a-f0-9]{8}(-[a-f0-9]{4}){3}-[a-f0-9]{12}$'
  THEN RAISE EXCEPTION 'OWN_RECEIPT_CONTRACT_INVALID';END IF;
 END IF;
 RETURN jsonb_build_object('version','own-receipt-print-facts.v1','frozenAt',frozen_at,'units',units_value,
  'employmentStartDate',start_value,'classification',jsonb_build_object('categoryCode',category_value,'categoryLabel',NULL,'jobTitle',job_title_value,'positionStructure',NULL),
  'paymentDestination',destination,'sources',jsonb_build_object('runId',record_value->>'runId','payloadSha256',record_value->>'payloadSha256',
  'inputSha256',record_value->>'inputSha256','resultSha256',record_value->>'resultSha256','employmentEvidenceSha256',employment_hash,
  'bankConfigurationVersion',bank_fact->'bankConfigurationVersion','bankApprovalId',bank_fact->'bankApprovalId'));
END $v6facts$;

ROLLBACK;
