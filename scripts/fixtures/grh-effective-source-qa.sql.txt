CREATE TEMP TABLE qa_checks(name text PRIMARY KEY);
CREATE FUNCTION pg_temp.ok(value boolean,label text) RETURNS void LANGUAGE plpgsql AS $fn$
BEGIN IF value IS DISTINCT FROM true THEN RAISE EXCEPTION 'QA failed: %',label; END IF;
 INSERT INTO qa_checks VALUES(label); END $fn$;
CREATE FUNCTION pg_temp.denied(statement text,expected text,label text) RETURNS void LANGUAGE plpgsql AS $fn$
DECLARE caught text; BEGIN
 BEGIN EXECUTE statement; EXCEPTION WHEN OTHERS THEN caught:=SQLERRM; END;
 IF caught IS NULL OR position(expected IN caught)=0 THEN RAISE EXCEPTION 'QA failed: % / expected % / got %',label,expected,caught; END IF;
 INSERT INTO qa_checks VALUES(label); END $fn$;
