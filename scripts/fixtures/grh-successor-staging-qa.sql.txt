CREATE TEMP TABLE stage_changes AS
 SELECT coalesce(b.entity,a.entity) entity,coalesce(b.row_key,a.row_key) row_key,
 CASE WHEN b.row_key IS NULL THEN 'add' WHEN a.row_key IS NULL THEN 'remove' ELSE 'replace' END operation,b.record previous_record,a.record
 FROM stage_before b FULL JOIN stage_after a USING(entity,row_key) WHERE b.record IS DISTINCT FROM a.record;
CREATE FUNCTION pg_temp.stage_proof() RETURNS jsonb LANGUAGE sql AS $f$
 SELECT jsonb_object_agg(e,jsonb_build_object(
 'baseline',(SELECT jsonb_build_object('rows',count(*),'md5',md5(coalesce(string_agg(md5(row_key||record::text),'' ORDER BY row_key),''))) FROM stage_before WHERE entity=e),
 'candidate',(SELECT jsonb_build_object('rows',count(*),'md5',md5(coalesce(string_agg(md5(row_key||record::text),'' ORDER BY row_key),''))) FROM stage_after WHERE entity=e),
 'changes',(SELECT jsonb_build_object('add',count(*) FILTER(WHERE operation='add'),'remove',count(*) FILTER(WHERE operation='remove'),'replace',count(*) FILTER(WHERE operation='replace')) FROM stage_changes WHERE entity=e)))
 FROM unnest(grh_successor_entities_v1()) e;
$f$;
