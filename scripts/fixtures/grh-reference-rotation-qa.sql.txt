CREATE TABLE qa_baseline AS SELECT * FROM source_xref;
CREATE TABLE qa_run_baseline AS SELECT * FROM data_import_runs;
CREATE TABLE qa_batch_baseline AS SELECT * FROM source_import_batch;
SET LOCAL municontrol.promotion_import_run_id='30';

DO $cases$
DECLARE mode text; testcase record; changed bigint; after_first jsonb; after_second jsonb;
 mutation text; iteration integer; expected jsonb; observed jsonb; query_name text;
BEGIN
 FOR iteration IN 1..2 LOOP
  mode:=CASE iteration WHEN 1 THEN 'physical' ELSE 'revision' END;
  query_name:='rotate_'||mode;
  DELETE FROM source_xref; INSERT INTO source_xref SELECT * FROM qa_baseline;
  -- The revision path must read the view, not accidentally use the physical table.
  IF mode='revision' THEN INSERT INTO grh_employees VALUES(30,7,'999',9393); END IF;
  EXECUTE format('SELECT %I()',query_name) INTO changed;
  PERFORM assert_ok(changed=5,mode||': five eligible references close');
  FOR testcase IN SELECT * FROM qa_expected ORDER BY qa_case LOOP
   SELECT to_jsonb(b) INTO expected FROM qa_baseline b WHERE b.qa_case=testcase.qa_case;
   IF testcase.should_close THEN
    expected:=jsonb_set(expected,'{valid_to}',to_jsonb('2026-09-10 15:17:30+00'::timestamptz));
   END IF;
   SELECT to_jsonb(x) INTO observed FROM source_xref x WHERE x.qa_case=testcase.qa_case;
   PERFORM assert_ok(observed IS NOT DISTINCT FROM expected,mode||': '||testcase.qa_case);
  END LOOP;
  PERFORM assert_ok((SELECT count(*) FROM source_xref)=(SELECT count(*) FROM qa_baseline),mode||': no inserted or deleted references');
  PERFORM assert_ok(NOT EXISTS((SELECT * FROM data_import_runs EXCEPT ALL SELECT * FROM qa_run_baseline)
   UNION ALL(SELECT * FROM qa_run_baseline EXCEPT ALL SELECT * FROM data_import_runs)),mode||': runs unchanged');
  PERFORM assert_ok(NOT EXISTS((SELECT * FROM source_import_batch EXCEPT ALL SELECT * FROM qa_batch_baseline)
   UNION ALL(SELECT * FROM qa_batch_baseline EXCEPT ALL SELECT * FROM source_import_batch)),mode||': batches unchanged');
  SELECT jsonb_agg(to_jsonb(x) ORDER BY qa_case) INTO after_first FROM source_xref x;
  EXECUTE format('SELECT %I()',query_name) INTO changed;
  SELECT jsonb_agg(to_jsonb(x) ORDER BY qa_case) INTO after_second FROM source_xref x;
  PERFORM assert_ok(changed=0 AND after_first=after_second,mode||': replay preserves the full history');

  -- A valid empty run and a missing run cannot reuse another run's employees.
  DELETE FROM source_xref; INSERT INTO source_xref SELECT * FROM qa_baseline;
  PERFORM set_config('municontrol.promotion_import_run_id','40',true);
  EXECUTE format('SELECT %I()',query_name) INTO changed;
  PERFORM assert_ok(changed=0,mode||': exact empty run');
  PERFORM set_config('municontrol.promotion_import_run_id','999',true);
  EXECUTE format('SELECT %I()',query_name) INTO changed;
  PERFORM assert_ok(changed=0,mode||': missing run');
  PERFORM set_config('municontrol.promotion_import_run_id','30',true);

  -- Every candidate provenance predicate is evaluated by the actual UPDATE.
  FOREACH mutation IN ARRAY ARRAY[
   $m$UPDATE data_import_runs SET status='failed' WHERE id=30$m$,
   $m$UPDATE data_import_runs SET source_name='other_curated' WHERE id=30$m$,
   $m$UPDATE data_import_runs SET source_sha256=repeat('9',64) WHERE id=30$m$,
   $m$UPDATE source_import_batch SET source_database='other_database' WHERE legacy_import_run_id=30$m$,
   $m$UPDATE source_import_batch SET source_system='PERSONAS' WHERE legacy_import_run_id=30$m$,
   $m$UPDATE source_import_batch SET source_sha256=repeat('8',64) WHERE legacy_import_run_id=30$m$,
   $m$UPDATE source_import_batch SET legacy_import_run_id=NULL WHERE legacy_import_run_id=30$m$
  ] LOOP
   EXECUTE mutation;
   EXECUTE format('SELECT %I()',query_name) INTO changed;
   PERFORM assert_ok(changed=0,mode||': candidate provenance '||mutation);
   UPDATE data_import_runs r SET status=b.status,source_name=b.source_name,source_sha256=b.source_sha256 FROM qa_run_baseline b WHERE b.id=r.id;
   UPDATE source_import_batch s SET source_database=b.source_database,source_system=b.source_system,source_sha256=b.source_sha256,legacy_import_run_id=b.legacy_import_run_id FROM qa_batch_baseline b WHERE b.id=s.id;
  END LOOP;
  PERFORM assert_ok(NOT EXISTS((SELECT * FROM source_xref EXCEPT ALL SELECT * FROM qa_baseline)
   UNION ALL(SELECT * FROM qa_baseline EXCEPT ALL SELECT * FROM source_xref)),mode||': all no-op guards preserve full rows');
 END LOOP;
 PERFORM assert_ok((SELECT count(*) FROM qa_checks)=72,'both source paths and all cases executed');
END $cases$;
