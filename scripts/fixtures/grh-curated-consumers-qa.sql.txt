CREATE TEMP TABLE qa_checks(name text PRIMARY KEY);
CREATE FUNCTION pg_temp.ok(value boolean,label text) RETURNS void LANGUAGE plpgsql AS $f$
BEGIN IF value IS DISTINCT FROM true THEN RAISE EXCEPTION 'QA failed %',label; END IF; INSERT INTO qa_checks VALUES(label); END $f$;
CREATE FUNCTION pg_temp.denied(statement text,expected text,label text) RETURNS void LANGUAGE plpgsql AS $f$
DECLARE caught text; BEGIN BEGIN EXECUTE statement; EXCEPTION WHEN OTHERS THEN caught:=SQLERRM; END;
 IF caught IS NULL OR position(expected IN caught)=0 THEN RAISE EXCEPTION 'QA failed %: %',label,caught; END IF; INSERT INTO qa_checks VALUES(label); END $f$;
